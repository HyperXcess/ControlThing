()=>{
 const R=REACT,J=JSX,SOCKET=SOCKETSTORE,BUTTONS=BUTTONSTORE;
 const [open,setOpen]=R.useState(false),[data,setData]=R.useState(null),[notice,setNotice]=R.useState(''),[selected,setSelected]=R.useState(0),[pending,setPending]=R.useState(false),[tick,setTick]=R.useState(Date.now());
 const [dragY,setDragY]=R.useState(null),[dragX,setDragX]=R.useState(0),gesture=R.useRef(null),tileCount=R.useRef(4);
 const send=(type,payload)=>SOCKET.getState().send({app:'controlthing',type:'app_payload',payload:{type,payload}});
 const action=(command,value)=>{if(pending)return;setPending(true);setNotice('');send('mission-control',{command,value});};
 R.useEffect(()=>{const toggle=e=>setOpen(current=>e.detail==='show'?true:e.detail==='hide'?false:!current);window.addEventListener('hyper-mission',toggle);return()=>window.removeEventListener('hyper-mission',toggle)},[]);
 R.useEffect(()=>{if(!open)return;const remove=SOCKET.getState().addListener(message=>{if(message.app!=='controlthing')return;if(message.type==='mission'){const receivedAt=Date.now();setData({...message.payload,receivedAt,focusEndsAt:message.payload.focus&&message.payload.focus.running?receivedAt+message.payload.focus.remaining*1000:null});setTick(receivedAt);setPending(false);if(message.payload.notice)setNotice(message.payload.notice)}else if(message.type==='mission-notice'){setNotice(message.payload.message);setPending(false)}});send('mission-get');const timer=setInterval(()=>{send('mission-get');setTick(Date.now())},3000);return()=>{clearInterval(timer);remove()}},[open]);
 R.useEffect(()=>{if(!open)return;const timer=setInterval(()=>setTick(Date.now()),200);return()=>clearInterval(timer)},[open]);
 R.useEffect(()=>{if(!open)return;let last=0;const wheel=e=>{e.preventDefault();e.stopImmediatePropagation();if(Date.now()-last<90)return;last=Date.now();const delta=Math.abs(e.deltaX)>Math.abs(e.deltaY)?e.deltaX:e.deltaY;setSelected(i=>(i+(delta>0?1:tileCount.current-1))%tileCount.current)};const key=e=>{if(e.key==='Escape'){setOpen(false);return}if(e.key==='Enter'){e.preventDefault();e.stopImmediatePropagation();if(selected===1){BUTTONS.getState().executeAction({id:'open',source:'server',value:'agenda',enabled:true});setOpen(false)}else if(selected===2)action(data&&data.focus.running?'focus-pause':'focus-start');else if(selected===3)action('brightness',(data&&data.brightness>=100?25:Math.min(100,Math.floor((data&&data.brightness||25)/25)*25+25)))}};window.addEventListener('wheel',wheel,{capture:true,passive:false});window.addEventListener('keydown',key,true);return()=>{window.removeEventListener('wheel',wheel,true);window.removeEventListener('keydown',key,true)}},[open,selected,data,pending]);
 R.useEffect(()=>{
  const start=e=>{const t=e.touches&&e.touches[0];if(!t||(!open&&t.clientY>32))return;if(open&&e.target.closest&&e.target.closest('select,input'))return;gesture.current={x:t.clientX,y:t.clientY,at:Date.now(),axis:null,wasOpen:open};};
  const move=e=>{const g=gesture.current,t=e.touches&&e.touches[0];if(!g||!t)return;const dx=t.clientX-g.x,dy=t.clientY-g.y;if(!g.axis&&Math.max(Math.abs(dx),Math.abs(dy))>12)g.axis=Math.abs(dx)>Math.abs(dy)&&g.wasOpen?'x':'y';if(!g.axis)return;e.preventDefault();e.stopImmediatePropagation();if(g.axis==='y')setDragY(Math.max(-window.innerHeight,Math.min(0,(g.wasOpen?0:-window.innerHeight)+dy)));else setDragX(Math.max(-window.innerWidth/2,Math.min(window.innerWidth/2,dx)));};
  const end=e=>{const g=gesture.current;gesture.current=null;if(!g||!g.axis)return;e.preventDefault();e.stopImmediatePropagation();const t=e.changedTouches&&e.changedTouches[0];const dx=t?t.clientX-g.x:0,dy=t?t.clientY-g.y:0;if(g.axis==='y')setOpen(g.wasOpen?dy>-60:dy>60);else if(Math.abs(dx)>45){const count=Math.ceil(tileCount.current/2),page=Math.floor(selected/2);setSelected(((page+(dx<0?1:count-1))%count)*2);}setDragY(null);setDragX(0);};
  const cancel=()=>{gesture.current=null;setDragY(null);setDragX(0)};
  window.addEventListener('touchstart',start,{capture:true,passive:true});window.addEventListener('touchmove',move,{capture:true,passive:false});window.addEventListener('touchend',end,{capture:true,passive:false});window.addEventListener('touchcancel',cancel,true);
  return()=>{window.removeEventListener('touchstart',start,true);window.removeEventListener('touchmove',move,true);window.removeEventListener('touchend',end,true);window.removeEventListener('touchcancel',cancel,true)};
 },[open,selected]);
 const stats=data&&data.stats||{},focus=data&&data.focus||{},remaining=focus.running&&data.focusEndsAt?Math.max(0,Math.ceil((data.focusEndsAt-tick)/1000)):(focus.remaining||0),time=Math.floor(remaining/60)+':'+String(remaining%60).padStart(2,'0');
 const button=(label,command,value,extra={})=>J.jsx('button',{disabled:pending,onClick:()=>action(command,value),style:{border:0,borderRadius:'8px',background:'#303842',color:'#fff',padding:'8px 11px',fontSize:'15px',...extra},children:label});
 const tile=(i,title,children)=>J.jsxs('section',{onClick:()=>setSelected(i),style:{background:'#171c23',border:'2px solid '+(selected===i?'#c1e5b9':'#303842'),borderRadius:'15px',padding:'12px 15px',overflow:'hidden',display:'flex',flexDirection:'column',gap:'7px'},children:[J.jsx('div',{style:{fontSize:'12px',letterSpacing:'1.5px',color:'#aeb8c4'},children:title}),...children]});
 const next=data&&data.next,minutes=next?Math.ceil((new Date(next.start).getTime()-tick)/60000):0;
 const tiles=[
 tile(0,'MAC HEALTH',[J.jsxs('div',{style:{display:'flex',justifyContent:'space-between',alignItems:'baseline'},children:[J.jsx('span',{style:{fontSize:'31px'},children:(stats.cpu==null?'—':stats.cpu+'%')+' CPU'}),J.jsx('span',{style:{fontSize:'22px',color:'#c1e5b9'},children:stats.battery==null?'':stats.battery+'% 🔋'})]}),J.jsx('div',{style:{fontSize:'16px',color:'#b8c2cd'},children:stats.memoryAvailable==null?'Measuring Mac…':stats.memoryAvailable+'% memory available'}),J.jsx('div',{style:{fontSize:'14px',color:'#8c9aab'},children:stats.power||''})]),
 tile(1,'YOUR DAY',[J.jsx('div',{style:{fontSize:'23px',lineHeight:1.15,maxHeight:'55px',overflow:'hidden'},children:next?next.title:data?'No upcoming meetings':'Loading your day…'}),J.jsx('div',{style:{fontSize:'19px',color:'#c1e5b9'},children:next?(minutes>0?'In '+minutes+' min':'Happening now'):''}),J.jsx('div',{style:{fontSize:'14px',color:'#aeb8c4',overflow:'hidden',whiteSpace:'nowrap',textOverflow:'ellipsis'},children:next?next.calendar||'':''})]),
 tile(2,'FOCUS',[J.jsxs('div',{style:{display:'flex',justifyContent:'space-between',alignItems:'center'},children:[J.jsx('span',{style:{fontSize:'35px',fontVariantNumeric:'tabular-nums'},children:time}),J.jsx('span',{style:{fontSize:'14px',color:'#c1e5b9'},children:(focus.completed||0)+' finished today'})]}),J.jsxs('div',{style:{display:'flex',gap:'6px'},children:[button(focus.running?'Pause':'Start',focus.running?'focus-pause':'focus-start'),button('Reset','focus-reset'),button((focus.duration||1500)/60+' min','focus-duration',({25:50,50:5,5:25})[(focus.duration||1500)/60]||25)]})]),
 tile(3,'QUICK CONTROLS',[J.jsxs('div',{style:{display:'flex',gap:'7px'},children:[button('Screen '+(data&&data.brightness!=null?data.brightness+'%':'☀'),'brightness',(data&&data.brightness>=100?25:Math.min(100,Math.floor((data&&data.brightness||25)/25)*25+25))),J.jsx('button',{onClick:()=>data&&data.focusShortcut?action('shortcut'):setNotice('Add your focus Shortcut name in ControlThing → Settings.'),disabled:pending,style:{background:'#303842',border:0,color:'#fff',borderRadius:'8px',padding:'8px 10px',fontSize:'15px'},children:'Focus Shortcut'})]}),J.jsx('select',{value:data&&data.audio&&data.audio.current||'',disabled:pending,onChange:e=>action('audio',Number(e.target.value)),style:{width:'100%',background:'#252c34',color:'#fff',border:'1px solid #46515f',borderRadius:'8px',padding:'8px',fontSize:'15px'},children:data&&data.audio&&data.audio.devices.length?data.audio.devices.map(device=>J.jsx('option',{value:device.id,children:device.name},device.id)):J.jsx('option',{value:'',children:'Audio outputs unavailable'})})])];
 tileCount.current=tiles.length;
 const page=Math.floor(selected/2),pageCount=Math.ceil(tiles.length/2),pages=Array.from({length:pageCount},(_,i)=>tiles.slice(i*2,i*2+2));
 const changePage=index=>setSelected(((index+pageCount)%pageCount)*2);
 return J.jsxs(J.Fragment,{children:[!open&&J.jsx('div',{style:{position:'fixed',top:3,left:'calc(50% - 24px)',width:'48px',height:'4px',borderRadius:'4px',background:'#ffffff66',zIndex:45,pointerEvents:'none'}}),J.jsxs('div',{'aria-hidden':!open&&dragY==null,style:{position:'fixed',top:0,left:0,right:0,bottom:0,zIndex:60,background:'#090c10',color:'#fff',padding:'15px 20px',display:'flex',flexDirection:'column',gap:'11px',borderRadius:'0 0 18px 18px',boxShadow:'0 12px 30px #0008',pointerEvents:open||dragY!=null?'auto':'none',transform:dragY==null?(open?'translateY(0)':'translateY(-105%)'):'translateY('+dragY+'px)',transition:dragY==null?'transform 220ms cubic-bezier(.2,.8,.2,1)':'none',willChange:open||dragY!=null?'transform':'auto'},children:[
 J.jsxs('header',{style:{display:'flex',alignItems:'center',justifyContent:'space-between'},children:[J.jsx('strong',{style:{fontSize:'25px',fontWeight:500},children:'ControlThing'}),J.jsx('span',{style:{color:'#aeb8c4',fontSize:'16px'},children:page===0?'Overview':page===1?'Tools':'Page '+(page+1)}),J.jsx('button',{onClick:()=>setOpen(false),style:{background:'#252c34',border:0,borderRadius:'8px',color:'#fff',padding:'7px 14px',fontSize:'17px'},children:'Close ↑'})]}),
 J.jsx('div',{style:{flex:1,minHeight:0,overflow:'hidden'},children:J.jsx('div',{style:{height:'100%',display:'flex',transform:'translateX(calc(-'+page*100+'% + '+dragX+'px))',transition:dragX?'none':'transform 200ms cubic-bezier(.2,.8,.2,1)'},children:pages.map((items,i)=>J.jsx('div',{'aria-hidden':i!==page,style:{flex:'0 0 100%',width:'100%',height:'100%',display:'grid',gridTemplateColumns:'1fr 1fr',gap:'11px',padding:'2px',boxSizing:'border-box'},children:items},i))})}),
 J.jsxs('nav',{'aria-label':'Shade pages',style:{display:'flex',justifyContent:'center',alignItems:'center',gap:'12px',height:'25px'},children:[J.jsx('button',{onClick:()=>changePage(page-1),'aria-label':'Previous page',style:{background:'none',border:0,color:'#fff',fontSize:'22px'},children:'‹'}),...pages.map((_,i)=>J.jsx('button',{onClick:()=>changePage(i),'aria-label':'Page '+(i+1),'aria-current':i===page?'page':undefined,style:{border:0,padding:0,width:i===page?'24px':'8px',height:'8px',borderRadius:'8px',background:i===page?'#c1e5b9':'#4e5966',transition:'width 150ms'},children:''},i)),J.jsx('button',{onClick:()=>changePage(page+1),'aria-label':'Next page',style:{background:'none',border:0,color:'#fff',fontSize:'22px'},children:'›'})]}),
 J.jsx('footer',{style:{fontSize:'13px',color:notice?'#c1e5b9':'#8c9aab',minHeight:'17px'},children:notice||'Swipe sideways: pages · Swipe up: close · Dial: select'})]})]});
}
